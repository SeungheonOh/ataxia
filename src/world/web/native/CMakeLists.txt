cmake_minimum_required(VERSION 3.21)
project(ataxia_web LANGUAGES C CXX)
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Set CEF_ROOT to the pinned CEF distribution; scripts/fetch-cef prints this path.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_library(ataxia-web-native SHARED bridge.c)
target_compile_options(ataxia-web-native PRIVATE -O2 -Wall -Wextra -fvisibility=hidden)
target_link_libraries(ataxia-web-native GLESv2 EGL pthread)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GBM REQUIRED IMPORTED_TARGET gbm libdrm)
add_executable(ataxia-web-helper helper.cpp gpu.cpp)
target_include_directories(ataxia-web-helper PRIVATE "${CEF_ROOT}")
target_compile_definitions(ataxia-web-helper PRIVATE ${CEF_COMPILER_DEFINES})
target_compile_options(ataxia-web-helper PRIVATE -O2 -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
target_link_libraries(ataxia-web-helper libcef_dll_wrapper "${CEF_ROOT}/Release/libcef.so" pthread dl EGL GLESv2 PkgConfig::GBM)
set_target_properties(ataxia-web-helper PROPERTIES BUILD_RPATH "$ORIGIN;${CEF_ROOT}/Release")
# Runtime resources remain in the verified SDK; no browser runtime in libataxia.
file(GLOB runtime_files "${CEF_ROOT}/Resources/*" "${CEF_ROOT}/Release/*")
foreach(path IN LISTS runtime_files)
  get_filename_component(name "${path}" NAME)
  file(CREATE_LINK "${path}" "${CMAKE_CURRENT_BINARY_DIR}/${name}" SYMBOLIC)
endforeach()
